def code2type(code):
    
    if code<120:
        incident='Homicide'
    elif code<300:
        incident='Rape'
    elif code<400:
        incident='Robbery'
    elif code<500:
        incident='Aggravated Assault'
    elif code>=3000 and code<4000:
        incident='Hospital Cases'
    else:
        incident=None
    
    return incident
